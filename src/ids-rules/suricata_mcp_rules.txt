# ==========================================================
# 1. TLS SNI (weak signal)
# ==========================================================

alert tls any any -> any any (
    msg:"SIG_MCP_SNI_GENERIC";
    tls.sni;
    content:"mcp"; nocase;
    classtype:trojan-activity;
    sid:1000001;
    rev:2;
)

    msg:"SIG_MCP_SNI_AGENT";
    tls.sni;
    content:"agent"; nocase;
    classtype:trojan-activity;

    rev:2;
)

# ==========================================================
# 3. Small TLS request (LLM pattern)
# ==========================================================

alert tls any any -> any any (
    msg:"SIG_SMALL_TLS_REQ";
    flow:established,to_server;
    dsize:<300;
    threshold:type both, track by_src, count 40, seconds 10;
    classtype:attempted-recon;
    sid:1000020;
    rev:2;
)


# ==========================================================
# 4. TLS burst (agent loop)
# ==========================================================

alert tcp any any -> any 443 (
    msg:"SIG_TLS_BURST";
    flags:S;
    threshold:type both, track by_src, count 25, seconds 5;
    classtype:attempted-dos;
    sid:1000030;
    rev:2;
)


# ==========================================================
# 5. Long-lived session (streaming / agents)
# ==========================================================

alert tls any any -> any any (
    msg:"SIG_LONG_TLS_SESSION";
    flow:established;
    threshold:type both, track by_src, count 1, seconds 600;
    classtype:policy-violation;
    sid:1000040;
    rev:2;
)


# ==========================================================
# 6. Repetitive request size (tool loop)
# ==========================================================

alert tls any any -> any any (
    msg:"SIG_REPETITIVE_TLS_PATTERN";
    flow:established,to_server;
    dsize:200<>400;
    threshold:type both, track by_src, count 60, seconds 30;
    classtype:attempted-recon;
    sid:1000050;
    rev:2;
)
